\end{minipage}\hfill
\begin{minipage}[t]{0.47\textwidth}\centering
\includegraphics[width=\linewidth]{objects/circuit_discocat.pdf}

\vspace{2pt}
{\scriptsize\textbf{(d) \texttt{DisCoCat}.} Word states with an IQP ansatz; a grammar cup is a Bell effect (CNOT, $H$, post-select $0$), leaving one open sentence wire.}
\end{minipage}
