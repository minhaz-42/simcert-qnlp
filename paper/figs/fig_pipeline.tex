  q/.style={draw=blue!55!black, line width=0.7pt, rounded corners=3pt, minimum height=11mm,

  c/.style={draw=teal!60!black, line width=0.7pt, rounded corners=3pt, minimum height=11mm,

% ---- the real input sentence, shared by both tracks ----
\node[s] (x) {engineer writes program};
\node[below=0.5mm of x, font=\scriptsize, black!55] (xl) {\textsc{mc} test item, true label $1$};

% ---- top track: the trained quantum model ----
\node[n, right=6mm of x] (e1) {embed};
\node[q, right=5mm of e1] (qc) {quantum core, $n{=}4$\\[-1pt]$U_{\mathrm{enc}}(x)\,U(\theta)$};
\node[rdo, right=5mm of qc] (r1) {$\langle Z_0\rangle$\\$=-0.843$};
\node[n, right=5mm of r1] (h1) {head};
\node[right=4mm of h1] (y1) {$\hat y=1$};
\foreach \a/\b in {e1/qc,qc/r1,r1/h1,h1/y1} \draw[->] (\a)--(\b);

\node[n, below=17mm of e1] (e2) {embed};

\caption{\textsc{SimCert} at a glance, on a real audited example. A trained quantum-NLP
classifier (top) embeds a sentence, runs a quantum core that prepares
$|\psi_\theta(x)\rangle$ and reads out Pauli expectations, and applies a classical head to
obtain a label. \textsc{SimCert} replaces the quantum core with a classical
matrix-product-state surrogate of bond dimension $\chi$ (bottom) and asks how small $\chi$
can be before the recovered label changes. The sentence, the expectation values, the
fidelity and both labels are measured values from the reference classifier on \textsc{mc}
at seed $1$, not illustrations. They show the case the paper is about: at $\chi{=}1$ the
surrogate is a product state that overlaps the trained state by only $F{=}0.52$, yet it
returns the same label. The trained state carries entanglement, $\bar S = 0.82$ nats here,
and the decision does not use it.}
\label{fig:pipeline}
