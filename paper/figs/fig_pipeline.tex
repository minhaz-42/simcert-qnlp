% Fig 1: two-track schematic. Top = trained quantum model; bottom = classical MPS surrogate that
% replaces the quantum core; linked by an equivalence. Minimal, paper-friendly palette.
\begin{figure}[t]
\centering
\begin{tikzpicture}[
  font=\small, >=Latex,
  n/.style={draw=black!40, rounded corners=2pt, minimum height=8.5mm, minimum width=13mm, inner sep=2pt, fill=black!4, align=center},
  q/.style={draw=blue!55!black, line width=0.7pt, rounded corners=3pt, minimum height=10mm, minimum width=34mm, inner sep=2pt, fill=blue!9, align=center},
  c/.style={draw=teal!60!black, line width=0.7pt, rounded corners=3pt, minimum height=10mm, minimum width=34mm, inner sep=2pt, fill=teal!12, align=center},
  lbl/.style={black!55, font=\itshape\footnotesize, align=right},
]
% ---- top track: trained quantum model ----
\node (x1) {$x$};
\node[n, right=4mm of x1] (e1) {embed};
\node[q, right=5mm of e1] (qc) {quantum core\\[-1pt]$U_{\mathrm enc}(x)\,U(\theta)\to\langle P\rangle$};
\node[n, right=5mm of qc] (h1) {head};
\node[right=4mm of h1] (y1) {$\hat y$};
\foreach \a/\b in {x1/e1,e1/qc,qc/h1,h1/y1} \draw[->] (\a)--(\b);
\node[lbl, left=3mm of x1] {trained\\quantum\\model};
% ---- bottom track: classical surrogate ----
\node[below=16mm of x1] (x2) {$x$};
\node[n, right=4mm of x2] (e2) {embed};
\node[c, right=5mm of e2] (mp) {classical MPS surrogate\\[-1pt]bond dimension $\chi$};
\node[n, right=5mm of mp] (h2) {head};
\node[right=4mm of h2] (y2) {$\hat y^{(\chi)}$};
\foreach \a/\b in {x2/e2,e2/mp,mp/h2,h2/y2} \draw[->] (\a)--(\b);
\node[lbl, left=3mm of x2] {\textsc{SimCert}\\surrogate};
% ---- equivalence between the two cores ----
\draw[<->, thick, dashed, black!55] (qc) -- (mp);
\node[right=1.5mm, align=left, font=\scriptsize, black!70] at ($(qc)!0.5!(mp)$)
  {truncate to bond $\chi$,\\ then $\hat y^{(\chi)}\!=\hat y$ for $\chi^\star\!\le 2$};
\end{tikzpicture}
\caption{\textsc{SimCert} at a glance. A trained quantum-NLP classifier (top) embeds a sentence
$x$, runs a quantum core that prepares $|\psi_\theta(x)\rangle$ and reads out Pauli expectations,
and applies a classical head to obtain a label $\hat y$. \textsc{SimCert} replaces the quantum core
with a classical matrix-product-state surrogate of bond dimension $\chi$ (bottom) and asks how small
$\chi$ can be before the recovered label $\hat y^{(\chi)}$ changes. For every audited model the two
tracks agree already at $\chi^\star\le 2$.}
\label{fig:pipeline}
\end{figure}
