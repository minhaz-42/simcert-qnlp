% Fig: dequantization picture in Shin (2024) Fig 1(b) style. The trained model's expectation is a
% coefficient MPS (blue) contracted with basis functions (red triangles); truncating the MPS bond
% dimension to chi gives a classical MPS that reproduces the output (basis-equivalent).
\begin{figure}[t]
\centering
\begin{tikzpicture}[
  font=\small, >=Latex,
  mps/.style={draw=blue!55!black, thick, fill=blue!14, rounded corners=1.5pt, minimum size=6.5mm},
  cmps/.style={draw=teal!65!black, thick, fill=teal!16, rounded corners=1.5pt, minimum size=6.5mm},
  tri/.style={draw=red!60!black, fill=red!16, regular polygon, regular polygon sides=3, shape border rotate=-90, minimum size=6.5mm, inner sep=0pt},
  grp/.style={draw, dashed, rounded corners=3pt, inner sep=3.5pt},
]
\def\dy{0.92}
% ===== quantum side: coefficient MPS x basis functions =====
\node at (-1.35,0) {$\langle O\rangle_{\theta,x}=$};
\foreach \k in {0,1,2,3}{
  \node[mps]  (Q\k) at (0, -\dy*\k) {};
  \node[tri]  (T\k) at (1.4, -\dy*\k) {};
  \draw[thick, black!55] (Q\k) -- (T\k);          % contraction leg
}
\draw[blue!55!black, thick] (Q0)--(Q1) (Q1)--(Q2) (Q2)--(Q3);
\node[blue!55!black] at (-0.55,{-\dy*1.5}) {\scriptsize$\chi$};
\node[draw=none] at (0,-\dy*3-0.7) {\scriptsize$\vdots$};
% dashed group labels
\node[grp, fit=(Q0)(Q3), blue!55!black] (Qbox) {};
\node[blue!55!black, above=8pt of Qbox.north, xshift=-1mm] {\scriptsize coefficient MPS $\mathbf{C}^{q}(\theta)$};
\node[grp, fit=(T0)(T3), red!55!black] (Tbox) {};
\node[red!55!black, above=1pt of Tbox.north] {\scriptsize basis $\mathbf{T}(x)$};

% ===== basis-equivalence arrow =====
\node (sim) at (2.55,-1.4*\dy) {$\underset{\text{\scriptsize basis-equivalent}}{\sim}$};

% ===== classical side: bond-chi classical MPS x same basis =====
\begin{scope}[shift={(4.0,0)}]
  \foreach \k in {0,1,2,3}{
    \node[cmps] (C\k) at (0, -\dy*\k) {};
    \node[tri]  (U\k) at (1.4, -\dy*\k) {};
    \draw[thick, black!55] (C\k) -- (U\k);
  }
  \draw[teal!65!black, thick] (C0)--(C1) (C1)--(C2) (C2)--(C3);
  \node[draw=none] at (0,-\dy*3-0.7) {\scriptsize$\vdots$};
  \node[grp, fit=(C0)(C3)(U0)(U3), teal!55!black] (Cbox) {};
  \node[teal!55!black, above=1pt of Cbox.north, align=center] {\scriptsize classical MPS,\\[-2pt]\scriptsize bond $\le\chi$};
\end{scope}
\end{tikzpicture}
\caption{Following the dequantization picture of \citet{shin2024}, the trained model's expectation value is a \emph{coefficient} matrix product
state $\mathbf{C}^{q}(\theta)$ (blue) contracted with a fixed set of basis functions
$\mathbf{T}(x)$ (red), and its bond dimension caps the entanglement it can carry. If that bond
dimension is small, a classical MPS of the same size (teal) contracts with the same basis functions
to reproduce the output, so the model is basis-equivalent to a cheap classical one. \textsc{SimCert}
measures this bond dimension directly: it truncates the trained state to bond dimension $\chi$ and
finds the smallest $\chi$ that still recovers the predictions, which for every audited model is $1$
or $2$, against an exact-representation bound of $2^{\lfloor n/2\rfloor}$.}
\label{fig:tn}
\end{figure}
