% Fig 2: simulability hierarchy as a clean set-inclusion chain (CSIM subset QESIM subset QSIM).
\begin{figure}[t]
\centering
\begin{tikzpicture}[
  font=\small,
  box/.style={rounded corners=3pt, minimum height=15mm, minimum width=30mm, align=center, line width=0.8pt},
]
\node[box, draw=teal!55!black, fill=teal!12] (csim)
  {\textbf{CSIM}\\[3pt]\footnotesize classically\\[-1pt]\footnotesize simulable};
\node[box, draw=blue!50, fill=blue!7, right=12mm of csim] (qesim)
  {\textbf{QESIM}\\[3pt]\footnotesize classical up\\[-1pt]\footnotesize to data loading};
\node[box, draw=black!45, fill=black!4, right=12mm of qesim] (qsim)
  {\textbf{QSIM}\\[3pt]\footnotesize quantum-\\[-1pt]\footnotesize simulable};
\node at ($(csim.east)!0.5!(qesim.west)$) {\large$\subseteq$};
\node at ($(qesim.east)!0.5!(qsim.west)$) {\large$\subseteq$};
\node[below=7mm of csim, align=center, font=\footnotesize, black!70] (note)
  {$\bigstar$~\textsc{SimCert} certifies membership here\\ when the required bond dimension is small};
\draw[->, black!55] (note) -- (csim);
\end{tikzpicture}
\caption{The nested hierarchy of classical simulability. A classically simulable model (CSIM) is a
special case of one that is simulable up to a data-loading step (QESIM), which is a special case of
quantum-simulable (QSIM) \citep{cerezo2025,shin2024}. \textsc{SimCert} certifies membership in CSIM
whenever the trained state truncates to a small bond dimension, and we make no claim outside that
innermost class.}
\label{fig:venn}
\end{figure}
